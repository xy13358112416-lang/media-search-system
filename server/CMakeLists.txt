find_package(Qt6 REQUIRED COMPONENTS Core Network Sql)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SODIUM REQUIRED IMPORTED_TARGET libsodium)
add_library(warehouse_server_lib STATIC warehouse_server.cpp)
target_include_directories(warehouse_server_lib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${PROJECT_SOURCE_DIR}/include)
target_compile_features(warehouse_server_lib PUBLIC cxx_std_17)
target_link_libraries(warehouse_server_lib PUBLIC warehouse_protocol Qt6::Core Qt6::Network Qt6::Sql PkgConfig::SODIUM)
target_compile_options(warehouse_server_lib PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
add_executable(media_server main.cpp)
target_link_libraries(media_server PRIVATE warehouse_server_lib)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(warehouse_server_tests tests/warehouse_server_tests.cpp)
    set_target_properties(warehouse_server_tests PROPERTIES AUTOMOC ON)
    target_link_libraries(warehouse_server_tests PRIVATE warehouse_server_lib Qt6::Test)
    target_compile_options(warehouse_server_tests PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
    add_test(NAME warehouse_server COMMAND warehouse_server_tests)
endif()
