cmake_minimum_required(VERSION 3.16)
project(MediaBusiness LANGUAGES CXX)
include(CTest)
find_package(Qt6 REQUIRED COMPONENTS Core Test)
add_subdirectory(../retrieval retrieval-build)
add_library(media_business STATIC csvmediarepository.cpp medialibrary.cpp)
target_include_directories(media_business PUBLIC ../include)
target_compile_features(media_business PUBLIC cxx_std_17)
target_link_libraries(media_business PUBLIC Qt6::Core media_search)
add_executable(media_business_tests tests.cpp)
set_target_properties(media_business_tests PROPERTIES AUTOMOC ON)
target_compile_definitions(media_business_tests PRIVATE SAMPLE_CSV="${CMAKE_CURRENT_SOURCE_DIR}/../sample-data/data/media_records.csv")
target_link_libraries(media_business_tests PRIVATE media_business Qt6::Test)
add_test(NAME media_business_tests COMMAND media_business_tests)
